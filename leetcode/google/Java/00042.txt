方法一：三次线性扫描(动态规划求leftMax和rightMax) 1ms
    public int trap(int[] height) {
        int n = height.length, ans = 0;
        if(n==0) {
            return 0;
        }

        int[] leftMax = new int[n];
        leftMax[0] = height[0];
        for(int i=1;i<n;i++) {
            leftMax[i] = Math.max(leftMax[i-1],height[i]);
        }

        int[] rightMax = new int[n];
        rightMax[n-1] = height[n-1];
        for(int i=n-2;i>=0;i--) {
            rightMax[i] = Math.max(rightMax[i+1],height[i]);
        }

        for(int i=0;i<n;i++) {
            ans += Math.min(leftMax[i],rightMax[i]) - height[i];
        }
        return ans;
    }

方法二：双指针(对方法一的改进，将空间复杂度降到O(1)) 0ms 最优做法
    public int trap(int[] height) {
        int left=0, right=height.length-1;
        int leftMax=0, rightMax=0, ans=0;

        while(left<right) {
            leftMax = Math.max(leftMax,height[left]);
            rightMax = Math.max(rightMax,height[right]);

            if(leftMax<rightMax) {
                ans += leftMax-height[left];
                left++;
            } else {
                ans += rightMax-height[right];
                right--;
            }
        }

        return ans;
    }

方法三：单调栈，由于用到了栈容器，实际运行速度会比上面两种慢 6ms(使用Stack时)
①
    public int trap(int[] height) {
        Stack<Integer> sk = new Stack<>();
        int ans = 0;

        for(int i=0; i<height.length; i++) {
            while(!sk.empty() && height[sk.peek()]<height[i]) {
                int top = sk.pop();
                if(sk.empty()) {
                    break;
                }

                int left = sk.peek();
                int curWidth = i-left-1, curHeight = Math.min(height[i],height[left])-height[top];
                ans += curWidth*curHeight;
            }

            sk.push(i);
        }

        return ans;
    }
