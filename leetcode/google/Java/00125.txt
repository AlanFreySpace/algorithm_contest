//方法1：筛选+判断(单纯使用API)
class Solution {
    public boolean isPalindrome(String s) {
        StringBuffer sgood = new StringBuffer();
        int len = s.length();

        for(int i=0; i<len; i++) {
            char c = s.charAt(i);
            if(Character.isLetterOrDigit(c)) {
                sgood.append(Character.toLowerCase(c));
            }
        }

        StringBuffer sgood_rev = new StringBuffer(sgood).reverse();
        return sgood.toString().equals(sgood_rev.toString());
    }
}

注意：
        StringBuffer sgood_rev = new StringBuffer(sgood).reverse();
        return sgood.toString().equals(sgood_rev.toString());
不要写成：
        String sgood_rev = sgood.reverse().toString();
        return sgood.toString().equals(sgood_rev);
因为sgood.reverse()会把sgood本身的值改掉。

//方法2：在原字符串上使用双指针直接判断
class Solution {
    public boolean isPalindrome(String s) {
        int left=0, right=s.length()-1;

        while(left<right) {
            while(left<right && !check(s.charAt(left))) {
                left++;
            }
            while(left<right && !check(s.charAt(right))) {
                right--;
            }
	    //此处无需二次判断left<right，上面两个while结束最多是left==right，此时if判断相等，最终返回true，
	    //这是符合题意的
            if(s.charAt(left)!=s.charAt(right) && 
                s.charAt(left)!=(s.charAt(right)^32)) {
                return false;
            }
            left++;
            right--;
        }

        return true;
    }

    boolean check(char c) {
        return (c>='0'&&c<='9') || (c>='a'&&c<='z') || (c>='A'&&c<='Z');
    }
}

注意：
s.charAt(left)!=(s.charAt(right)^32)
不要写成：
s.charAt(left)!=s.charAt(right)^32，
否则Java会认为对s.charAt(left)!=s.charAt(right)的布尔值进行异或，而不是对s.charAt(right)的ASCII码值进行异或，从而报错。
